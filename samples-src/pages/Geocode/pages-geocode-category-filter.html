Geocodage avec plusieurs categories

Ancien Glacier du Puy Gris

var options = {
    query: "Ancien Glacier du Puy Gris",
    index: "PositionOfInterest",
    maximumResponses: 1,
    returnTrueGeometry: true,
    filters: {
        category: ["glacier, névé", "plan d'eau"],
        postcode: 73130
    }
};
Gp.Services.geocode(options);
